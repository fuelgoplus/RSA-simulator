                headHtml += `<th class="${alignClass} ${colClasses[i]}">${h}</th>`;
            });
            const headRow = document.getElementById('tableHeadRow');
            if (headRow) headRow.innerHTML = headHtml;

            renderManualInputs();
            updateTable();
            calculateTargetTDC();
        }

        function renderManualInputs() {

        function updateManualData(index, field, value) {
            manualData[index][field] = parseFloat(value) || 0;
            updateTable();
            calculateTargetTDC();
        }

        function syncGlobalToManual() {
            const vals = [salaryInp.value, rsaInp.value, stockInp.value, cashDistInp.value, exerciseInp.value];
            const simBaseEl = document.getElementById('simBaseInput');
            const simGrantEl = document.getElementById('simGrantInput');
            const simPriceEl = document.getElementById('simPriceInput');
            const simDivEl = document.getElementById('simDivInput');
            
            if (simBaseEl) simBaseEl.value = vals[0];
            if (simGrantEl) simGrantEl.value = vals[1];
            if (simPriceEl) simPriceEl.value = vals[2];
            if (simDivEl) simDivEl.value = vals[3];

            for(let i=0; i<=10; i++) {
                manualData[i].salary = parseFloat(vals[0]) || 0;
                manualData[i].grant = parseFloat(vals[1]) || 0;
                manualData[i].price = parseFloat(vals[2]) || 0;
                manualData[i].cashDist = parseFloat(vals[3]) || 0;
                manualData[i].exercise = parseFloat(vals[4]) || 0;
            }
            renderManualInputs();
            updateTable();
            calculateTargetTDC();
        }

        function updateTable() {
            const t = i18n[currentLang];
            const years = parseInt(slider.value);
            const yearValEl = document.getElementById('yearVal');
            if (yearValEl) yearValEl.innerText = years;
            const tableTitleEl = document.getElementById('tableTitle');
            if (tableTitleEl) tableTitleEl.innerText = `${t.tableTitlePrefix}${years}${t.tableTitleSuffix}`;
            
            let html = "";
            let sums = { sal:0, shr:0, val:0, soldQty:0, soldVal:0, div:0, tdc:0 };
            let cumGrant = 0;
            let cumVested = 0;
            let cumSold = 0;

            for (let y = 0; y <= years; y++) {
                let vShr = 0; let cHtml = "";
                const cur = manualData[y];
                cumGrant += cur.grant;

                for (let g = 0; g <= y; g++) {
                    let idx = y - g; let p = 0;
                    if (idx === 1) p = 0.2; else if (idx === 2) p = 0.3; else if (idx === 3) p = 0.5;
                    if (p > 0) {
                        const amt = manualData[g].grant;
                        const v = amt * p; vShr += v;
                        const vestVal = v * cur.price;
                        const shareLabel = currentLang === 'zh' ? '股' : 'sh';
                        const valLabel = currentLang === 'zh' ? '值' : 'Val';
                        cHtml += `<span class="calc-line">G${g}: <span class="font-bold">${Math.round(amt).toLocaleString()}</span>×${Math.round(p*100)}%=<span class="font-bold" style="color: var(--matrix-primary1)">${Math.round(v).toLocaleString()}</span>${shareLabel} <span class="text-[9px] opacity-80 ml-1 font-mono">(${valLabel}: ${Math.round(vestVal).toLocaleString()})</span></span>`;
                    }
                }
                
                if (cHtml === "") {
                    cHtml = `<span class="calc-line opacity-50">${currentLang === 'zh' ? '期初授予，尚無歸屬' : 'Initial Grant (No Vest)'}</span>`;
                }

                cumVested += vShr;

                let actualSell = Math.max(0, Math.min(cur.exercise || 0, cumVested - cumSold));
                cumSold += actualSell;
                let sellVal = actualSell * cur.price;

                const eqV = vShr * cur.price;
                const heldShares = Math.max(0, cumGrant - cumSold);
                const divV = heldShares * cur.cashDist;
                const tdc = cur.salary + eqV + divV;
                
                sums.tdc += tdc;
                sums.sal += cur.salary; sums.shr += vShr; sums.val += eqV; sums.div += divV;
                sums.soldQty += actualSell; sums.soldVal += sellVal;

                let avg = sums.tdc / (y + 1);

                html += `
                    <tr class="hover:bg-black/5 transition-colors">
                        <td class="font-bold">Y${y}</td>
                        <td class="numeric">${Math.round(cur.salary).toLocaleString()}</td>
                        <td class="numeric font-bold" style="color: var(--matrix-primary1)">${Math.round(vShr).toLocaleString()}</td>
                        <td class="col-calc-cell">${cHtml}</td>
                        <td class="numeric">${Math.round(eqV).toLocaleString()}</td>
                        <td class="numeric font-bold" style="color: var(--matrix-negative)">${Math.round(actualSell).toLocaleString()}</td>
                        <td class="numeric font-bold" style="color: var(--matrix-primary1)">
                            ${Math.round(divV).toLocaleString()}<br>
                            <span class="text-[9px] opacity-70 block font-mono font-normal tracking-tighter">Hold: ${Math.round(heldShares).toLocaleString()}</span>
                        </td>
                        <td class="numeric font-bold" style="background-color: var(--matrix-bg); opacity: 0.9">${Math.round(tdc).toLocaleString()}</td>
                        <td class="numeric font-bold" style="background-color: var(--matrix-panel)">${Math.round(avg).toLocaleString()}</td>
                    </tr>
                `;
            }
            const tableBody = document.getElementById('tableBody');
            if (tableBody) tableBody.innerHTML = html;
            const tableFoot = document.getElementById('tableFoot');
            if (tableFoot) {
                tableFoot.innerHTML = `
                    <tr>
                        <td class="py-2">${t.sumAccumulated}</td>
                        <td class="numeric">${Math.round(sums.sal).toLocaleString()}</td>
                        <td class="numeric">${Math.round(sums.shr).toLocaleString()}</td>
                        <td class="col-calc-cell">-</td>
                        <td class="numeric">${Math.round(sums.val).toLocaleString()}</td>
                        <td class="numeric" style="color: var(--matrix-negative)">${Math.round(sums.soldQty).toLocaleString()}</td>
                        <td class="numeric">${Math.round(sums.div).toLocaleString()}</td>
                        <td class="numeric" style="color: var(--matrix-accent)">${Math.round(sums.tdc).toLocaleString()}</td>
                        <td class="numeric" style="color: var(--matrix-accent)">${Math.round(sums.tdc/(years + 1)).toLocaleString()}</td>
                    </tr>
                `;
            }
        }

        function calculateTargetTDC() {
            const target = parseFloat(document.getElementById('simTargetInput')?.value) || 0;
            const base = parseFloat(document.getElementById('simBaseInput')?.value) || 0;
            const grant = parseFloat(document.getElementById('simGrantInput')?.value) || 0;
            const price = parseFloat(document.getElementById('simPriceInput')?.value) || 0;
            const div = parseFloat(document.getElementById('simDivInput')?.value) || 0;

            let cumGrant = 0;
            let sumsTDC = 0;
            let hitYear = -1;
            let hitAvg = 0;

            // Forward simulate up to 30 years to find intersection
            for(let y = 0; y <= 30; y++) {
                cumGrant += grant;
                let vShr = 0;
                for (let g = 0; g <= y; g++) {
                    let idx = y - g; let p = 0;
                    if (idx === 1) p = 0.2; else if (idx === 2) p = 0.3; else if (idx === 3) p = 0.5;
                    if (p > 0) vShr += grant * p;
                }
                const eqV = vShr * price;
                const divV = cumGrant * div; 
                const tdc = base + eqV + divV;
                sumsTDC += tdc;
                const avg = sumsTDC / (y + 1);

                if (y > 0 && avg >= target) {
                    hitYear = y;
                    hitAvg = avg;
                    break;
                }
            }

            if (hitYear !== -1 && target > 0) {
                safeSetText('simResultYear', `Y${hitYear}`);
                safeSetText('simResultAvg', `NT$ ${Math.round(hitAvg).toLocaleString()}`);
            } else if (target > 0) {
                safeSetText('simResultYear', `>30 Yrs`);
                safeSetText('simResultAvg', `-`);
            } else {
                safeSetText('simResultYear', `-`);
                safeSetText('simResultAvg', `NT$ 0`);
            }
        }

        slider.addEventListener('input', () => { renderManualInputs(); updateTable(); });
        [salaryInp, rsaInp, stockInp, cashDistInp, exerciseInp].forEach(el => el.addEventListener('input', syncGlobalToManual));
        window.onload = () => { initData(); updateUI(); };
